def _anti_debug():
    """Anti-debugging protection"""
    try:
        if sys.gettrace() is not None:
            os._exit(0)
    except:
        pass

    try:
        import ctypes
        if os.name == 'nt' and ctypes.windll.kernel32.IsDebuggerPresent():
            os._exit(0)
    except:
        pass

    try:
        # Check for common debugging tools
        debugger_files = ['/usr/bin/gdb', '/usr/bin/strace', '/usr/bin/ltrace']
        for f in debugger_files:
            if os.path.exists(f):
                # Don't exit, just note it
                pass
    except:
        pass

def _decrypt_multi_layer(data, keys):
    """Decrypt multiple layers"""
    decrypted = data

    # Decrypt in reverse order
    for key in reversed(keys):
        # XOR decrypt
        xored = bytes([decrypted[i] ^ key[i % len(key)] for i in range(len(decrypted))])
        # Decompress
        decrypted = zlib.decompress(xored)

    return decrypted

def _execute_protected():
    """Execute the protected code"""
    try:
        # Anti-debugging
        _anti_debug()

        # Decode from Base85
        _decoded = base64.b85decode(_DATA)

        # Multi-layer decryption
        _decrypted = _decrypt_multi_layer(_decoded, _KEYS)

        # Convert to source code
        _source = _decrypted.decode('utf-8')

        # Execute
        exec(compile(_source, '<protected>', 'exec'), globals(), globals())

    except SystemExit:
        raise
    except Exception as e:
        print(f"Execution error: {e}", file=sys.stderr)
        sys.exit(1)
