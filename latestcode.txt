def check_network_connectivity():
    print('[+] Checking network connectivity...')
    try:
        socket.create_connection(("8.8.8.8", 53), timeout=3)
        print('[+] Network connectivity confirmed')
        return True
    except OSError:
        print('[!] Primary DNS check failed, trying alternative...')
        try:
            socket.create_connection(("1.1.1.1", 53), timeout=3)
            print('[+] Network connectivity confirmed via alternative DNS')
            return True
        except OSError:
            print('[-] No network connectivity detected')
            return False

def http_get_with_retry(url, timeout=5, max_retries=MAX_RETRIES, retry_delay=RETRY_DELAY):
    if not url:
        print('[-] No URL provided for HTTP request')
        return None
    
    print(f'[+] Fetching data from: {url}')
    
    for attempt in range(max_retries):
        try:
            print(f'[+] Attempt {attempt + 1}/{max_retries}')
            response = requests.get(url, timeout=timeout, headers={"User-Agent": "Mozilla/5.0"})
            
            if response.status_code == 404:
                print('[!] Resource not found (404)')
                return '404: Not Found'
            elif response.status_code == 403:
                print('[!] Access forbidden (403), retrying...')
                time.sleep(retry_delay)
                continue
            elif response.status_code == 429:
                print('[!] Rate limited (429), waiting longer...')
                time.sleep(retry_delay * 2)
                continue
            elif response.status_code != 200:
                print(f'[!] Unexpected status code: {response.status_code}')
                time.sleep(retry_delay)
                continue
            
            print('[+] Data fetched successfully')
            return response.text
        except requests.exceptions.ReadTimeout:
            print(f'[-] Timeout error on attempt {attempt + 1}')
            time.sleep(retry_delay)
            continue
        except requests.exceptions.ConnectionError:
            print(f'[-] Connection error on attempt {attempt + 1}')
            time.sleep(retry_delay)
            continue
        except requests.exceptions.RequestException as e:
            print(f'[-] Request exception: {e}')
            time.sleep(retry_delay)
            continue
        except Exception as e:
            print(f'[-] Unexpected error: {e}')
            time.sleep(retry_delay)
            continue
    
    print('[+] Falling back to urllib...')
    try:
        req = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0"})
        with urllib.request.urlopen(req, timeout=timeout) as resp:
            data = resp.read().decode('utf-8')
            print('[+] Data fetched successfully via urllib')
            return data
    except urllib.error.HTTPError as e:
        if e.code == 404:
            print('[!] Resource not found (404)')
            return '404: Not Found'
        print(f'[-] HTTP error: {e.code}')
        return None
    except Exception as e:
        print(f'[-] Urllib error: {e}')
        return None
    
    print('[-] All retry attempts failed')
    return None

def get_public_ip():
    print('[+] Retrieving public IP address...')
    if not check_network_connectivity():
        return "No internet connection"
    
    result = http_get_with_retry("https://api.ipify.org/", timeout=10)
    if result and result != '404: Not Found':
        ip = result.strip()
        print(f'[+] Public IP retrieved: {ip}')
        return ip
    
    print('[-] Failed to retrieve public IP')
    return "Unable to retrieve IP"

def auto_update():
    global CODE_URL
    print('[+] Starting auto-update process...')
    
    if not check_network_connectivity():
        print('[-] No internet connection for update')
        return False
    
    print(f'[+] Fetching latest code from: {CODE_URL}')
    for attempt in range(MAX_RETRIES):
        remote_code = http_get_with_retry(CODE_URL, timeout=10)
        if remote_code and not remote_code.startswith('404'):
            print('[+] Latest code fetched successfully')
            break
        print(f'[!] Failed to fetch code (attempt {attempt + 1}/{MAX_RETRIES})')
        time.sleep(RETRY_DELAY)
    
    if not remote_code or remote_code.startswith('404'):
        print('[-] Failed to fetch latest code')
        if CODE_URL != DEFAULT_CODE_URL:
            print('[+] Resetting to default CODE_URL...')
            CODE_URL = DEFAULT_CODE_URL
            print('[+] CODE_URL reset to default')
            print('[+] Retrying with default URL...')
            remote_code = http_get_with_retry(CODE_URL, timeout=10)
            if not remote_code or remote_code.startswith('404'):
                print('[-] Failed to fetch code from default URL')
                return False
        else:
            print('[-] Repository appears to be down')
            return False
    
    current_file = os.path.abspath(__file__)
    print(f'[+] Writing updated code to: {current_file}')
    try:
        with open(current_file, 'w', encoding='utf-8') as f:
            f.write(remote_code)
        print('[+] Update completed successfully')
        return True
    except PermissionError:
        print('[-] Permission denied when writing file')
        print('[!] Run script with appropriate permissions')
        return False
    except Exception as e:
        print(f'[-] Failed to write update: {e}')
        return False

def resolve_version_url():
    global VERSION_URL
    print('[!] Encountered Error')
    time.sleep(0.1)
    print('[+] Resolving...')
    time.sleep(0.2)
    print('[+] Identified Error : VERSION_URL is empty or invalid')
    print('[+] Attempting to resolve...')
    time.sleep(0.2)
    if VERSION_URL != DEFAULT_VERSION_URL:
        VERSION_URL = DEFAULT_VERSION_URL
        print('[+] Reset VERSION_URL to default')
        print('[+] Retrying...')
        time.sleep(0.2)
        return True
    else:
        print('[-] VERSION_URL already set to default')
        print('[-] Error is external or script is corrupt')
        return False

def fetch_latest_version():
    global VERSION_URL
    print('[+] Fetching latest version...')
    
    if not VERSION_URL:
        if not resolve_version_url():
            return None, False
    
    print('[+] Checking network connectivity...')
    if not check_network_connectivity():
        return None, False
    
    print(f'[+] Fetching from: {VERSION_URL}')
    latestver = http_get_with_retry(VERSION_URL, timeout=10)
    
    if not latestver:
        print('[!] Encountered Error')
        time.sleep(0.1)
        print('[+] Resolving...')
        time.sleep(0.2)
        print('[+] Identified Error : Network issue or server timeout')
        print('[+] Attempting to resolve...')
        time.sleep(0.2)
        print('[+] Retrying with extended timeout...')
        latestver = http_get_with_retry(VERSION_URL, timeout=15)
        if not latestver:
            print('[-] Resolve failed : Network unavailable')
            print('[-] This is an external error, check internet connection')
            return None, False
    
    if '404' in latestver or 'not found' in latestver.lower():
        print('[!] Encountered Error')
        time.sleep(0.1)
        print('[+] Resolving...')
        time.sleep(0.2)
        print('[+] Identified Error : Repository not found')
        print('[+] Attempting to resolve...')
        time.sleep(0.2)
        if VERSION_URL != DEFAULT_VERSION_URL:
            VERSION_URL = DEFAULT_VERSION_URL
            print('[+] Reset VERSION_URL to default')
            print('[+] Retrying...')
            time.sleep(0.2)
            latestver = http_get_with_retry(VERSION_URL, timeout=10)
            if not latestver or '404' in latestver:
                print('[-] Resolve failed : Repository is down')
                print('[-] This is an external error, cannot be fixed by script')
                return None, False
        else:
            print('[-] Resolve failed : Repository is down')
            print('[-] This is an external error, cannot be fixed by script')
            return None, False
    
    print(f'[+] Latest version retrieved: {latestver.strip()}')
    return latestver.strip(), True

def password_generator():
    clear()
    art()
    print('[+] Password Generator')
    
    try:
        print('[+] Reading parameters...')
        try:
            length = int(input('[>] Enter password length (minimum 4): '))
            if length < 4:
                print('[!] Password length must be at least 4 characters')
                length = 4
                print('[+] Using minimum length of 4')
        except ValueError:
            print('[!] Invalid length, using default of 12')
            length = 12
        
        use_upper = input('[>] Include uppercase? (y/n): ').lower() == 'y'
        use_digits = input('[>] Include digits? (y/n): ').lower() == 'y'
        use_special = input('[>] Include special chars? (y/n): ').lower() == 'y'
        
        print('[+] Generating password...')
        chars = string.ascii_lowercase
        if use_upper:
            chars += string.ascii_uppercase
        if use_digits:
            chars += string.digits
        if use_special:
            chars += string.punctuation
        
        password = ''.join(secrets.choice(chars) for _ in range(length))
        print(f'[+] Generated password: {password}')
    except Exception as e:
        print(f'[-] Error: {e}')
    
    input('\n[>] Press enter to continue...')

def hash_cracker():
    clear()
    art()
    print('[+] Hash Cracker')
    print('[^] 1) MD5')
    print('[^] 2) SHA1')
    print('[^] 3) SHA256')
    
    try:
        algo = input('[>] Select algorithm: ')
        print('[+] Reading hash...')
        hash_to_crack = input('[>] Enter hash to crack: ')
        print('[+] Reading wordlist path...')
        wordlist_path = input('[>] Enter wordlist path: ')
        
        if not os.path.exists(wordlist_path):
            print('[-] Wordlist not found')
            input('\n[>] Press enter to continue...')
            return
        
        print('[+] Starting hash cracking...')
        with open(wordlist_path, 'r', encoding='utf-8', errors='ignore') as f:
            for word in f:
                word = word.strip()
                if algo == '1':
                    hashed = hashlib.md5(word.encode()).hexdigest()
                elif algo == '2':
                    hashed = hashlib.sha1(word.encode()).hexdigest()
                elif algo == '3':
                    hashed = hashlib.sha256(word.encode()).hexdigest()
                else:
                    print('[!] Invalid algorithm')
                    return
                
                if hashed == hash_to_crack:
                    print(f'[+] Found: {word}')
                    input('\n[>] Press enter to continue...')
                    return
        
        print('[-] Password not found in wordlist')
    except Exception as e:
        print(f'[-] Error: {e}')
    
    input('\n[>] Press enter to continue...')

def dns_records():

        print(f'[+] Fetching DNS records for {domain}...')
        
        record_types = ['A', 'AAAA', 'MX', 'NS', 'TXT', 'CNAME', 'SOA']
        for record_type in record_types:
            try:
                print(f'[+] Fetching {record_type} records...')
                answers = dns.resolver.resolve(domain, record_type)

            except Exception as e:
                print(f'[!] Error fetching {record_type} records: {e}')
    except Exception as e:
        print(f'[-] Error: {e}')
    
    input('\n[>] Press enter to continue...')

def dns_zone_transfer():
    clear()
    art()
    print('[+] DNS Zone Transfer')
    
    try:
        print('[+] Reading domain name...')
        domain = input('[>] Enter domain: ')
        
        print('[+] Finding name servers...')
        try:
            ns_records = dns.resolver.resolve(domain, 'NS')
            name_servers = [str(ns) for ns in ns_records]
            print(f'[+] Found name servers: {name_servers}')
        except Exception as e:
            print(f'[-] Failed to find name servers: {e}')
            input('\n[>] Press enter to continue...')
            return
        
        for ns in name_servers:
            print(f'[+] Attempting zone transfer from {ns}...')
            try:
                zone = dns.zone.from_xfr(dns.query.xfr(ns, domain))
                print(f'[+] Zone transfer successful from {ns}')
                for name, node in zone.nodes.items():
                    for rdataset in node.rdatasets:
                        print(f'[+] {name} {rdataset}')
                break
            except Exception as e:
                print(f'[!] Zone transfer failed from {ns}: {e}')
        else:
            print('[-] Zone transfer failed from all name servers')
    except Exception as e:
        print(f'[-] Error: {e}')
    
    input('\n[>] Press enter to continue...')

        prefix = sha1_hash[:5]

        url = f'https://api.pwnedpasswords.com/range/{prefix}'
        
        print('[+] Checking password against breach database...')
        response = http_get_with_retry(url, timeout=10)
        if response:
            print('[+] Analyzing results...')
            for line in response.split('\n'):

        else:

def port_scanner():
    clear()
    art()
    print('[+] Port Scanner')
    
    try:
        print('[+] Reading scan parameters...')
        host = input('[>] Enter host: ')
        start_port = int(input('[>] Start port: ') or '1')
        end_port = int(input('[>] End port: ') or '1024')
        
        print(f'[+] Scanning {host} ports {start_port}-{end_port}...')
        print('[+] Using threaded scanning for speed...')
        
        open_ports = []
        lock = threading.Lock()
        
        def scan_port(port):
            try:
                sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
                sock.settimeout(1)
                result = sock.connect_ex((host, port))
                if result == 0:
                    with lock:
                        open_ports.append(port)
                        print(f'[+] Port {port}: OPEN')
                sock.close()
            except:
                pass
        
        with ThreadPoolExecutor(max_workers=50) as executor:
            futures = [executor.submit(scan_port, port) for port in range(start_port, end_port + 1)]
            for future in as_completed(futures):
                pass
        
        if not open_ports:
            print('[-] No open ports found')
        else:
            print(f'[+] Found {len(open_ports)} open ports')
    except Exception as e:
        print(f'[-] Error: {e}')
    
    input('\n[>] Press enter to continue...')

def ping_sweep():
    clear()
    art()
    print('[+] Ping Sweep')
    
    try:
        print('[+] Reading network address...')
        network = input('[>] Enter network (e.g., 192.168.1): ')
        print(f'[+] Sweeping {network}.0/24...')
        
        hosts_up = []
        lock = threading.Lock()
        
        def ping_host(i):
            host = f'{network}.{i}'
            try:
                if os.name == 'nt':
                    result = subprocess.run(['ping', '-n', '1', '-w', '1000', host], capture_output=True, text=True)
                else:
                    result = subprocess.run(['ping', '-c', '1', '-W', '1', host], capture_output=True, text=True)
                if result.returncode == 0:
                    with lock:
                        hosts_up.append(host)
                        print(f'[+] Host up: {host}')
            except:
                pass
        
        with ThreadPoolExecutor(max_workers=50) as executor:
            futures = [executor.submit(ping_host, i) for i in range(1, 255)]
            for future in as_completed(futures):
                pass
        
        print(f'[+] Found {len(hosts_up)} hosts up')
    except Exception as e:
        print(f'[-] Error: {e}')
    
    input('\n[>] Press enter to continue...')

def password_strength_checker():
    clear()
    art()
    print('[+] Password Strength Checker')
    
    try:
        print('[+] Reading password...')
        password = input('[>] Enter password: ')
        print('[+] Analyzing password strength...')
        
        score = 0
        length = len(password)
        
        if length >= 8:
            score += 1
        if length >= 12:
            score += 1
        if length >= 16:
            score += 1
        
        if re.search(r'[a-z]', password):
            score += 1
        if re.search(r'[A-Z]', password):
            score += 1
        if re.search(r'[0-9]', password):
            score += 1
        if re.search(r'[!@#$%^&*(),.?":{}|<>]', password):
            score += 1
        
        if score >= 7:
            print('[+] Strength: Very Strong')
        elif score >= 5:
            print('[+] Strength: Strong')
        elif score >= 3:
            print('[!] Strength: Medium')
        else:
            print('[-] Strength: Weak')
    except Exception as e:
        print(f'[-] Error: {e}')
    
    input('\n[>] Press enter to continue...')
